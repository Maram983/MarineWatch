/* MarineWatch - My Activities */ document.addEventListener("mw:ready", () => { const activities = { upcoming: JSON.parse(localStorage.getItem("joinedActivities")) || [], completed: [ { id: "A-090", title: "Jeddah Corniche Cleanup", city: "Jeddah", date: "2026-06-15", hours: 4, status: "completed" }, { id: "A-088", title: "KAUST Awareness Workshop", city: "Thuwal", date: "2026-06-02", hours: 3, status: "completed" } ], cancelled: [ { id: "A-077", title: "Jazan Mangrove Day", city: "Jazan", date: "2026-05-01", hours: 0, status: "cancelled" } ] }; const badges = { confirmed: 'Confirmed', completed: 'Completed', cancelled: 'Cancelled' }; function renderActivities(tab) { const list = activities[tab]; const grid = document.getElementById("grid"); if (!list.length) { grid.innerHTML = `

No activities found

You don't have any activities in this section.

`; return; } grid.innerHTML = list.map(activity => `
${MW.formatDateString(activity.date || "TBD")}
${badges[activity.status]}

${activity.title}

${activity.city} ${activity.hours || 3}h
Details ${ activity.status === "confirmed" ? ` ` : "" }
`).join(""); document.querySelectorAll(".leave-btn").forEach(button => { button.addEventListener("click", () => { let joined = JSON.parse(localStorage.getItem("joinedActivities")) || []; joined = joined.filter(item => item.id !== button.dataset.id); localStorage.setItem( "joinedActivities", JSON.stringify(joined) ); activities.upcoming = joined; renderActivities("upcoming"); mwToast("Activity removed", "success"); }); }); } const tabs = document.querySelectorAll("#tabs button"); tabs.forEach(button => { button.addEventListener("click", () => { tabs.forEach(tab => tab.classList.remove("active")); button.classList.add("active"); renderActivities(button.dataset.tab); }); }); renderActivities("upcoming"); });
